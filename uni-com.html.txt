      Career Connect
    </h1>
    <p class="text-slate-400 mt-2 text-sm">大学と企業をつなぐ、スマート相互検索</p>
  </header>

  <!-- Search App Wrapper -->
  <main class="w-full max-w-2xl glass-card rounded-2xl p-6 shadow-2xl">
    
    <!-- Mode Switcher -->
    <div class="flex bg-slate-800/80 p-1 rounded-xl mb-6">
      <button id="btnUni" onclick="setMode('uni')" class="flex-1 py-2.5 rounded-lg text-sm font-semibold transition-all duration-200 bg-gradient-to-r from-blue-500 to-purple-600 text-white shadow-lg">
        🎓 大学名から探す
      </button>
      <button id="btnComp" onclick="setMode('comp')" class="flex-1 py-2.5 rounded-lg text-sm font-semibold transition-all duration-200 text-slate-400 hover:text-white">
        🏢 企業名から探す
      </button>
    </div>

    <!-- Search Bar -->
    <div class="relative mb-6">
      <input type="text" id="searchInput" oninput="handleSearch()" placeholder="大学名を入力 (例: 東京大学)..." 
        class="w-full bg-slate-900/90 text-white placeholder-slate-500 text-sm rounded-xl pl-11 pr-4 py-3.5 border border-slate-700/80 focus:outline-none focus:border-purple-500 focus:ring-2 focus:ring-purple-500/20 transition-all">
      <svg class="w-5 h-5 text-slate-500 absolute left-3.5 top-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
        <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z"></path>
      </svg>
    </div>

    <!-- Results Area -->
    <div id="results" class="space-y-3"></div>
  </main>

  <script>
    const database = [
      { uni: "東京大学", comp: "Google Japan", count: 42, ratio: "12%" },
      { uni: "東京大学", comp: "ソニー", count: 38, ratio: "10%" },
      { uni: "東京大学", comp: "トヨタ自動車", count: 35, ratio: "9%" },
      { uni: "慶應義塾大学", comp: "三菱UFJ銀行", count: 85, ratio: "18%" },
      { uni: "慶應義塾大学", comp: "リクルート", count: 62, ratio: "14%" },
      { uni: "早稲田大学", comp: "NTTデータ", count: 78, ratio: "16%" },
      { uni: "早稲田大学", comp: "楽天グループ", count: 54, ratio: "11%" },
    ];

    let currentMode = 'uni';

    function setMode(mode) {
      currentMode = mode;
      const btnUni = document.getElementById('btnUni');
      const btnComp = document.getElementById('btnComp');
      const input = document.getElementById('searchInput');

      if(mode === 'uni') {
        btnUni.className = "flex-1 py-2.5 rounded-lg text-sm font-semibold transition-all duration-200 bg-gradient-to-r from-blue-500 to-purple-600 text-white shadow-lg";
        btnComp.className = "flex-1 py-2.5 rounded-lg text-sm font-semibold transition-all duration-200 text-slate-400 hover:text-white";
        input.placeholder = "大学名を入力 (例: 東京大学)...";
      } else {
        btnComp.className = "flex-1 py-2.5 rounded-lg text-sm font-semibold transition-all duration-200 bg-gradient-to-r from-purple-500 to-pink-600 text-white shadow-lg";
        btnUni.className = "flex-1 py-2.5 rounded-lg text-sm font-semibold transition-all duration-200 text-slate-400 hover:text-white";
        input.placeholder = "企業名を入力 (例: ソニー)...";
      }
      handleSearch();
    }

    function handleSearch() {
      const query = document.getElementById('searchInput').value.trim().toLowerCase();
      const resultsContainer = document.getElementById('results');
      resultsContainer.innerHTML = '';

      if(!query) {
        resultsContainer.innerHTML = `<p class="text-center text-slate-500 text-sm py-8">キーワードを入力して検索してください</p>`;
        return;
      }

      const filtered = database.filter(item => {
        const target = currentMode === 'uni' ? item.uni : item.comp;
        return target.toLowerCase().includes(query);
      });

      if(filtered.length === 0) {
        resultsContainer.innerHTML = `<p class="text-center text-slate-500 text-sm py-8">該当するデータが見つかりませんでした</p>`;
        return;
      }

      filtered.forEach(item => {
        const title = currentMode === 'uni' ? item.comp : item.uni;
        const sub = currentMode === 'uni' ? item.uni : item.comp;
        
        const card = document.createElement('div');
        card.className = "p-4 rounded-xl bg-slate-800/50 border border-slate-700/50 hover:border-slate-600 transition-all flex items-center justify-between";
        card.innerHTML = `
          <div>
            <div class="text-xs text-purple-400 font-medium">${sub}</div>
            <div class="text-base font-bold text-white mt-0.5">${title}</div>
          </div>
          <div class="text-right">
            <span class="text-sm font-bold text-blue-400">${item.count}名</span>
            <span class="text-xs text-slate-500 block">実績比率 ${item.ratio}</span>
          </div>
        `;
        resultsContainer.appendChild(card);
      });
    }

    handleSearch();
  </script>
